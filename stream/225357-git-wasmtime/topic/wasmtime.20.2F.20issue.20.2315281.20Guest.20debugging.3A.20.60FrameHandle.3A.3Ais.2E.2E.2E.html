impl DebugHandler for OnTrap {
    type Data = ();
    async fn handle(&self, mut store: StoreContextMut<'_, ()>, event: DebugEvent<'_>) {
        if let DebugEvent::Trap(_) = event {
            *self.0.lock().unwrap() = store.debug_exit_frames().next();
        }
    }
}

fn block_on<F: Future>(f: F) -> F::Output {
    let mut f = std::pin::pin!(f);
    let mut cx = Context::from_waker(Waker::noop());
    loop {
        if let Poll::Ready(v) = f.as_mut().poll(&mut cx) {

        }
    }

$ cargo run
trap: is_valid = true
trap: local 0 = Ok(I32(4))
component: is_valid = true
component: local 0 = Ok(I32(-340539887))

Both handles should be invalid, because their frames are gone by then. Instead, local reads stale stack memory. With a funcref local, the same stale read builds a Func from a guest-written pointer and segfaults. The same happens for handles taken in DebugEvent::HostcallError and DebugEvent::Exception handlers.
