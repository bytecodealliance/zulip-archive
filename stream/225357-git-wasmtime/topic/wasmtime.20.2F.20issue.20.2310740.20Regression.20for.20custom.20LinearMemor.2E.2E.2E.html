[package]
name = "wasmtime-repro-10740"
version = "0.1.0"
edition = "2024"

[dependencies]
wasmtime = "=49.0.3"
rustix = { version = "1", features = ["mm"] }

//! Usage: repro-10740-v49 <memory_init_cow: true|false>

use rustix::mm::{MapFlags, MprotectFlags, ProtFlags, mmap_anonymous, mprotect, munmap};
use std::ptr::null_mut;
use std::sync::Arc;
use wasmtime::*;

struct CustomMmapMemory {
    base: usize,
    mapped: usize,
    size: usize,
    capacity: usize,
}

impl CustomMmapMemory {
    /// Reserves `reserve + guard` bytes as `PROT_NONE` and makes `size` bytes read/write, growing in place up to `reserve`.
    fn new(minimum: usize, reserve: usize, guard: usize) -> Result<Self> {
        let mapped = reserve + guard;
        let base =
            unsafe { mmap_anonymous(null_mut(), mapped, ProtFlags::empty(), MapFlags::PRIVATE)? };
        if minimum > 0 {
            unsafe { mprotect(base, minimum, MprotectFlags::READ | MprotectFlags::WRITE)? };
        }
        Ok(Self {
            base: base as usize,
            mapped,
            size: minimum,
            capacity: reserve,
        })
    }
}

impl Drop for CustomMmapMemory {
    fn drop(&mut self) {
        unsafe { munmap(self.base as *mut _, self.mapped).expect("munmap failed") };
    }
}

unsafe impl LinearMemory for CustomMmapMemory {
    fn byte_size(&self) -> usize {
        self.size
    }

    fn byte_capacity(&self) -> usize {
        self.capacity
    }

    fn grow_to(&mut self, new_size: usize) -> Result<()> {
        if new_size > self.capacity {
            bail!("cannot grow beyond the reservation");
        }
        unsafe {
            mprotect(
                (self.base + self.size) as *mut _,
                new_size - self.size,
                MprotectFlags::READ | MprotectFlags::WRITE,
            )?;
        }
        self.size = new_size;
        Ok(())
    }

    fn as_ptr(&self) -> *mut u8 {
        self.base as *mut u8
    }
}

struct CustomMemoryCreator;

unsafe impl MemoryCreator for CustomMemoryCreator {
    fn new_memory(
        &self,
        _ty: MemoryType,
        minimum: usize,
        maximum: Option<usize>,
        reserved_size_in_bytes: Option<usize>,
        guard_size_in_bytes: usize,
    ) -> std::result::Result<Box<dyn LinearMemory>, String> {
        let reserve = reserved_size_in_bytes
            .or(maximum)
            .unwrap_or(1 << 30)
            .max(minimum);
        Ok(Box::new(
            CustomMmapMemory::new(minimum, reserve, guard_size_in_bytes)
                .map_err(|r| r.to_string())?,
        ))
    }
}

fn main() -> Result<()> {
    let cow: bool = match std::env::args().nth(1).as_deref() {
        Some("true") => true,
        Some("false") => false,
        _ => {
            eprintln!("usage: wasmtime-repro-10740 <memory_init_cow: true|false>");
            std::process::exit(1);
        }
    };

    let mut config = Config::new();
    config.with_host_memory(Arc::new(CustomMemoryCreator));
    config.memory_init_cow(cow);
    let engine = Engine::new(&config)?;

    // The active data segment is what produces a CoW memory image.
    let module = Module::new(
        &engine,
        r#"
        (module
          (memory (export "mem") 1)
          (data (i32.const 0) "hello")
          (func (export "load") (param i32) (result i32)
            (i32.load8_u (local.get 0))))
        "#,
    )?;

    let mut store = Store::new(&engine, ());
    println!("memory_init_cow = {cow}; instantiating...");
    let instance = Instance::new(&mut store, &module, &[])?;
    let load = instance.get_typed_func::<i32, i32>(&mut store, "load")?;
    let byte = load.call(&mut store, 1)?;
    println!(
        "load(1) = {byte:?} ({:?}), expected 'e'",
        byte as u8 as char
    );
    assert_eq!(byte, i32::from(b'e'));
    Ok(())
}

use rustix::mm::{MapFlags, MprotectFlags, ProtFlags, mmap_anonymous, mprotect, munmap};
use std::ptr::null_mut;
use std::sync::Arc;
use wasmtime::*;

struct CustomMmapMemory {
    base: usize,
    mapped: usize,
    size: usize,
    capacity: usize,
}

impl CustomMmapMemory {
    /// Reserves `reserve + guard` bytes as `PROT_NONE` and makes `size` bytes read/write, growing in place up to `reserve`.
    fn new(minimum: usize, reserve: usize, guard: usize) -> Result<Self> {
        let mapped = reserve + guard;
        let base =
            unsafe { mmap_anonymous(null_mut(), mapped, ProtFlags::empty(), MapFlags::PRIVATE)? };
        if minimum > 0 {
            unsafe { mprotect(base, minimum, MprotectFlags::READ | MprotectFlags::WRITE)? };
        }
        Ok(Self {
            base: base as usize,
            mapped,
            size: minimum,
            capacity: reserve,
        })
    }
}

impl Drop for CustomMmapMemory {
    fn drop(&mut self) {
        unsafe { munmap(self.base as *mut _, self.mapped).expect("munmap failed") };
    }
}

unsafe impl LinearMemory for CustomMmapMemory {
    fn byte_size(&self) -> usize {
        self.size
    }

    fn byte_capacity(&self) -> usize {
        self.capacity
    }

    fn grow_to(&mut self, new_size: usize) -> Result<()> {
        if new_size > self.capacity {
            bail!("cannot grow beyond the reservation");
        }
        unsafe {
            mprotect(
                (self.base + self.size) as *mut _,
                new_size - self.size,
                MprotectFlags::READ | MprotectFlags::WRITE,
            )?;
        }
        self.size = new_size;
        Ok(())
    }

    fn as_ptr(&self) -> *mut u8 {
        self.base as *mut u8
    }
}
