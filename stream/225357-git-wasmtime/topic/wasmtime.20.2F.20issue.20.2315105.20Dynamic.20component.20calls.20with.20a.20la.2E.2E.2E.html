Test Case

component.wat: a component exporting echo: func(data: list<u8>) -> list<u8>, which hands the list back, and len: func(data: list<u8>) -> u32, which only lowers it.

;; `echo` hands the list back; `len` returns its length (isolates lowering).
;; `realloc` always returns offset 16 (one allocation per call), growing memory as needed.
(component
  (core module $m
    (memory (export "memory") 1)
    (func (export "realloc") (param i32 i32 i32 i32) (result i32) (local $pages i32)
      (local.set $pages (i32.shr_u (i32.add (local.get 3) (i32.const 65551)) (i32.const 16)))
      (if (i32.gt_u (local.get $pages) (memory.size))
        (then (drop (memory.grow (i32.sub (local.get $pages) (memory.size))))))
      (i32.const 16))
    (func (export "echo") (param i32 i32) (result i32)
      (i32.store (i32.const 0) (local.get 0))
      (i32.store (i32.const 4) (local.get 1))
      (i32.const 0))
    (func (export "len") (param i32 i32) (result i32) (local.get 1)))
  (core instance $i (instantiate $m))
  (func (export "echo") (param "data" (list u8)) (result (list u8))
    (canon lift (core func $i "echo") (memory (core memory $i "memory")) (realloc (core func $i "realloc"))))
  (func (export "len") (param "data" (list u8)) (result u32)
    (canon lift (core func $i "len") (memory (core memory $i "memory")) (realloc (core func $i "realloc")))))

The host program calls both exports with 1, 16 and 64 MiB lists, dynamically (Func::call_async with Val::List, as an embedder that only learns signatures at runtime must) and through TypedFunc<(Vec<u8>,), _>, and prints each call's time and how far peak RSS rose during it.

use std::time::Instant;
use wasmtime::component::{Component, Linker, Val};
use wasmtime::{Config, Engine, Store};

const WAT: &str = include_str!("../component.wat");

fn kib(field: &str) -> u64 {
    let s = std::fs::read_to_string("/proc/self/status").unwrap();
    let l = s.lines().find(|l| l.starts_with(field)).unwrap();
    l.split_whitespace().nth(1).unwrap().parse().unwrap()
}

// Times one call and reports how far RSS peaked above the RSS right before it.
macro_rules! bench {
    ($label:expr, $mib:expr, $call:expr) => {{
        std::fs::write("/proc/self/clear_refs", "5").unwrap(); // reset VmHWM
        let base = kib("VmRSS:");
        let t = Instant::now();
        let r = $call;
        let dt = t.elapsed();
        println!("{:<14} {:>3} MiB {:>10.1?}  peak RSS +{:>5} MiB", $label, $mib, dt, kib("VmHWM:").saturating_sub(base) >> 10);
        r
    }};
}

#[tokio::main(flavor = "current_thread")]
async fn main() -> wasmtime::Result<()> {
    let concurrent = std::env::var_os("NO_CONCURRENCY").is_none();
    let mut config = Config::new();
    config.concurrency_support(concurrent).wasm_component_model_async(concurrent);
    let engine = Engine::new(&config)?;
    let component = Component::new(&engine, WAT)?;
    let mut store = Store::new(&engine, ());
    // The default (128 MiB) rejects dynamic lifts of list<u8> above ~2.7 MiB, since
    // each element is charged size_of::<Val>() = 48 bytes.
    store.set_hostcall_fuel(usize::MAX);
    let instance = Linker::new(&engine).instantiate_async(&mut store, &component).await?;
    let echo = instance.get_func(&mut store, "echo").unwrap();
    let len = instance.get_func(&mut store, "len").unwrap();
    let echo_t = echo.typed::<(Vec<u8>,), (Vec<u8>,)>(&store)?;
    let len_t = len.typed::<(Vec<u8>,), (u32,)>(&store)?;
    println!("concurrency_support = {concurrent}, size_of::<Val>() = {}", size_of::<Val>());

    for mib in [1usize, 16, 64] {
        let bytes = vec![7u8; mib << 20];
        let params = [Val::List(bytes.iter().map(|b| Val::U8(*b)).collect())];
        let mut out = [Val::Bool(false)];

        bench!("dynamic len", mib, len.call_async(&mut store, &params, &mut out).await?);
        assert!(matches!(out[0], Val::U32(n) if n as usize == bytes.len()));
        bench!("dynamic echo", mib, echo.call_async(&mut store, &params, &mut out).await?);
        assert!(matches!(&out[0], Val::List(v) if v.len() == bytes.len()));
        drop((params, out));

        let arg = (bytes.clone(),);
        let (n,) = bench!("typed len", mib, len_t.call_async(&mut store, arg).await?);
        assert_eq!(n as usize, bytes.len());
        let arg = (bytes.clone(),);
        let (v,) = bench!("typed echo", mib, echo_t.call_async(&mut store, arg).await?);
        assert_eq!(v, bytes);
    }
    Ok(())
}

Steps to Reproduce

  • Create the project above (Cargo.toml, src/main.rs, component.wat).
  • Run cargo run --release (Linux: the peak-RSS column reads /proc/self/clear_refs and VmHWM).
  • Run NO_CONCURRENCY=1 cargo run --release to compare with Config::concurrency_support(false).

Expected Results

A dynamic call moving a list<u8> costs within a small factor of the typed call, as both copy the same bytes between host and guest: time and host memory proportional to the list's bytes, and the default hostcall fuel admitting the same list sizes on both paths.

Actual Results

At 16-64 MiB the dynamic calls are 100-200x slower than the typed ones, and host memory rises by about 48 bytes per element (size_of::<Val>() == 48): a 16 MiB list<u8> is a 768 MiB Vec<Val>.

Concurrency support on (the default):

call 1 MiB 16 MiB 64 MiB peak RSS rise at 16 / 64 MiB
dynamic len (lower only) 25-51 ms 390-523 ms 2.2-2.9 s +782 / +3119 MiB
typed len 0.15 ms 2.5-2.7 ms 6.2-12.4 ms ~0 / ~0 MiB
dynamic echo (lower + lift) 46-104 ms 722-837 ms 4.4-5.8 s +768 / +3084 MiB
typed echo 0.4-2.2 ms 5.7-7.8 ms 19-28 ms +16 / +63 MiB

Concurrency support off (NO_CONCURRENCY=1, same binary):

call 1 MiB 16 MiB 64 MiB peak RSS rise at 16 / 64 MiB
dynamic len 11.6 ms 175-189 ms 752-767 ms +15 / +48 MiB
dynamic echo 33.4 ms 497-538 ms 2.4-2.5 s +768 / +3084 MiB
typed len / echo 0.12 / 0.33-0.40 ms 2.6-3.5 / 6.0-7.7 ms 8.1-8.5 / 24-26 ms ~0 / +16, +63 MiB
  • With concurrency support on, the dynamic len (lowering only) takes 2-4x as long as with it off, and its peak RSS rises by an extra 48 B per element: the arguments are deep-copied (params.to_vec(), see below).
  • With the default hostcall fuel (128 MiB), the dynamic echo fails at 3 MiB: too much data is being copied between the host and the guest: fuel allocated for hostcalls has been exhausted. The dynamic lift is charged size_of::<Val>() per element, the typed one size_of::<u8>(), which stays within the default up to 128 MiB.

Expected Results

A dynamic call moving a list<u8> costs within a small factor of the typed call, as both copy the same bytes between host and guest: time and host memory proportional to the list's bytes, and the default hostcall fuel admitting the same list sizes on both paths.

Actual Results

At 16-64 MiB the dynamic calls are 100-200x slower than the typed ones, and host memory rises by about 48 bytes per element (size_of::<Val>() == 48): a 16 MiB list<u8> is a 768 MiB Vec<Val>.

  • With the default hostcall fuel (128 MiB), the dynamic echo fails at 3 MiB: too much data is being copied between the host and the guest: fuel allocated for hostcalls has been exhausted. The dynamic lift is charged size_of::<Val>() per element, the typed one size_of::<u8>(), which stays within the default up to 128 MiB.
